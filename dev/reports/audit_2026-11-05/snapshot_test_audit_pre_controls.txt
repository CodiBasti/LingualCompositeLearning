"""Checks that matter for an audit: statistical calibration and provenance."""
import hashlib
import json
import pathlib
import unittest
import numpy as np
import analyze

HERE=pathlib.Path(__file__).resolve().parent
ROOT=HERE.parents[2]

class AuditMethods(unittest.TestCase):
    def test_student_t_reference_quantiles(self):
        self.assertAlmostEqual(analyze.ibeta(.5,.5,.5),.5,places=12)
        self.assertAlmostEqual(analyze.ibeta(4.5,.5,9/(9+2.2621571628**2)),.05,places=10)
        self.assertAlmostEqual(analyze.ibeta(49.5,.5,99/(99+1.9842169515**2)),.05,places=9)

    def test_small_sample_exact_randomization(self):
        result=analyze.stats([1,1,1,1],[0,0,0,0])
        self.assertEqual(result['permutation_p_two_sided'],2/16)
        self.assertEqual(result['ci95'],[1.,1.])
        self.assertEqual(analyze.stats([0,0],[0,0])['permutation_p_two_sided'],1)

    def test_contrast_direction_and_reproducibility(self):
        a=list(np.linspace(0,1,40));b=list(reversed(a))
        self.assertEqual(analyze.stats(a,b),analyze.stats(a,b))
        self.assertAlmostEqual(analyze.stats(a,b)['mean_difference'],-analyze.stats(b,a)['mean_difference'])
        self.assertEqual(analyze.stats(a,b)['permutation_p_two_sided'],analyze.stats(b,a)['permutation_p_two_sided'])
        with self.assertRaises(ValueError):analyze.stats([1],[0,1])
        with self.assertRaises(ValueError):analyze.stats([1,0],[0,1],['a'])

    def test_holm_monotonic_adjustment(self):
        values={str(i):{'n':40,'permutation_p_two_sided':p} for i,p in enumerate([.01,.04,.03,1])}
        analyze.holm(values)
        self.assertEqual([round(values[str(i)]['permutation_p_holm'],5) for i in range(4)],[.04,.09,.09,1])

    def test_cluster_resampling_preserves_family(self):
        result=analyze.stats([1,1,0,0],[0,0,0,0],['a','a','b','b'])
        self.assertEqual(result['cluster_sensitivity']['clusters'],2)
        self.assertEqual(result['cluster_sensitivity']['ci95'],[0.,1.])

    def test_rounding_not_used_for_raw_recall(self):
        s={'scenario_id':'unit','domain':'test','expected_concept':'x','required_invariants':['a','b','c']}
        r={'scenario_id':'unit','plan':'','eval':{'matched_invariants':['a'],'missed_invariants':['b','c'],'recall':.333}}
        summary,rr,_=analyze.tabulate([r],[s])
        self.assertEqual(rr[0]['recall'],1/3)
        self.assertEqual(summary['macro_recall'],1/3)

    def test_duplicate_and_missing_ids_are_reported(self):
        s={'scenario_id':'unit','domain':'test','required_invariants':[]}
        other={**s,'scenario_id':'missing'}
        r={'scenario_id':'unit','plan':'','eval':{'matched_invariants':[],'missed_invariants':[]}}
        summary,_,_=analyze.tabulate([r,r],[s,other])
        self.assertEqual(summary['duplicates'],['unit'])
        self.assertEqual(summary['missing_ids'],['missing'])

    def test_freeze_content_has_not_changed(self):
        for freeze in ['execution_freeze.json','execution_freeze_v2.json']:
            if not (HERE/freeze).exists():continue
            frozen=json.loads((HERE/freeze).read_text(encoding='utf-8'))
            for name,digest in frozen['files'].items():
                path=ROOT/name
                if freeze=='execution_freeze.json' and path.name=='gpu_audit.py':path=HERE/'gpu_audit_v1.py'
                self.assertEqual(hashlib.sha256(path.read_bytes()).hexdigest(),digest,name)

    def test_holdout_complete_and_does_not_overlap(self):
        tasks=json.loads((HERE/'holdout.json').read_text(encoding='utf-8'))
        old=json.loads((ROOT/'dev/data/benchmarks/benchmark_v2.json').read_text(encoding='utf-8'))
        self.assertEqual(len(tasks),40)
        self.assertEqual(len({s['scenario_id'] for s in tasks}),40)
        self.assertFalse({s['scenario_id'] for s in tasks}&{s['scenario_id'] for s in old})
        self.assertFalse({s['context'] for s in tasks}&{s['context'] for s in old})
        self.assertEqual(set(__import__('collections').Counter(s['domain'] for s in tasks).values()),{10})

if __name__=='__main__':unittest.main(verbosity=2)
